def generate_tick(sym):
    """Generates realistic market data. Replace this block with Dhan API calls for live data."""
    cfg = UNIVERSE[sym]
    hist = st.session_state.history[sym]
    open_p = st.session_state.open_prem[sym]
    
    if not hist:
        spot = cfg['spot'] + random.uniform(-10, 10)
        prem = open_p
    else:
        last = hist[-1]
        # Random walk for spot
        spot = last['spot'] + random.gauss(0, cfg['spot'] * 0.0005)
        # Premium: mean reversion + theta decay + occasional volatility spikes
        spike_chance = 0.03 # 3% chance of a volatility event per tick
        if random.random() < spike_chance:
            prem_change = last['prem'] * random.uniform(0.04, 0.08) * random.choice([-1, 1])
        else:
            prem_change = random.gauss(0, cfg['base_prem'] * 0.005) - (cfg['base_prem'] * 0.0002) # slight theta decay
            
        prem = max(cfg['base_prem'] * 0.3, last['prem'] + prem_change)

    ce = prem * (0.45 + random.uniform(0, 0.1))
    pe = prem - ce
    pct_chg = ((prem - open_p) / open_p) * 100
    atm_strike = round(spot / cfg['step']) * cfg['step']
    
    tick = {
        'time': datetime.now().strftime("%H:%M:%S"),
        'spot': round(spot, 2),
        'prem': round(prem, 2),
        'ce': round(ce, 2),
        'pe': round(pe, 2),
        'pct_chg': round(pct_chg, 2),
        'strike': int(atm_strike)
    }
    
    st.session_state.history[sym].append(tick)
    if len(st.session_state.history[sym]) > 60:
        st.session_state.history[sym].pop(0)
        
    # Spike Detection (>= 5%)
    if abs(pct_chg) >= 5.0:
        # Prevent duplicate alerts for the same spike
        recent_alert = any(a['sym'] == sym and (datetime.now().timestamp() - a['ts']) < 30 for a in st.session_state.alerts)
        if not recent_alert:
            st.session_state.alerts.insert(0, {
                'ts': datetime.now().timestamp(),
                'time': tick['time'],
                'sym': sym,
                'pct': pct_chg,
                'prem': tick['prem'],
                'spot': tick['spot']
            })
            if len(st.session_state.alerts) > 20:
                st.session_state.alerts.pop()

                if isinstance(val, (int, float)) and abs(val) >= threshold:

                elif isinstance(val, (int, float)) and val > 0:
                    return 'color: #22c55e;'
                elif isinstance(val, (int, float)) and val < 0:
                    return 'color: #ef4444;'
                return ''
